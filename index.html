descargar_video.py 
# Descargar Videos de YouTube — 100% Gratis
import yt_dlp

def descargar_video(url, calidad="best"):
    formatos = {
        "best":  "bestvideo+bestaudio/best",
        "1080": "bestvideo[height<=1080]+bestaudio",
        "720":  "bestvideo[height<=720]+bestaudio",
        "480":  "bestvideo[height<=480]+bestaudio",
        "mp3":  "bestaudio/best",
    }

    opts = {
        "format": formatos.get(calidad),
        "merge_output_format": "mp4",
    }

    with yt_dlp.YoutubeDL(opts) as ydl:
        ydl.download([url])
    print("✅ ¡Descarga completada!")
